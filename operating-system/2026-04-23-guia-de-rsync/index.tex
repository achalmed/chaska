\textbf{¿Qué es rsync?}

\textbf{rsync} (remote sync) es una herramienta de sincronización de
archivos y directorios que:

\begin{itemize}
\tightlist
\item
  \textbf{Sincroniza eficientemente:} Solo transfiere las diferencias
  entre archivos
\item
  \textbf{Preserva metadatos:} Permisos, propietarios, timestamps,
  enlaces simbólicos
\item
  \textbf{Funciona local y remotamente:} A través de SSH u otros
  protocolos
\item
  \textbf{Es incremental:} Solo copia lo que ha cambiado
\item
  \textbf{Permite recuperación:} Puede reanudar transferencias
  interrumpidas
\end{itemize}

\subsection{Ventajas de rsync}\label{ventajas-de-rsync}

\textbf{Comparado con \texttt{cp}:}

\begin{itemize}
\tightlist
\item
  Solo copia archivos modificados
\item
  Preserva todos los atributos
\item
  Muestra progreso
\item
  Puede excluir archivos/directorios
\item
  Funciona sobre red
\end{itemize}

\textbf{Comparado con \texttt{scp}:}

\begin{itemize}
\tightlist
\item
  Mucho más rápido (solo diferencias)
\item
  Puede reanudar transferencias
\item
  Mejor manejo de errores
\item
  Más opciones de filtrado
\end{itemize}

\textbf{Comparado con otras herramientas:}

\begin{itemize}
\tightlist
\item
  Más eficiente que tar para backups incrementales
\item
  Más flexible que soluciones comerciales
\item
  Multiplataforma (Linux, macOS, Windows)
\end{itemize}

\section{Conceptos Fundamentales}\label{conceptos-fundamentales}

\subsection{Algoritmo de rsync}\label{algoritmo-de-rsync}

rsync usa un algoritmo que:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Divide archivos en bloques}
\item
  \textbf{Calcula checksums} de cada bloque
\item
  \textbf{Compara checksums} entre origen y destino
\item
  \textbf{Transfiere solo bloques diferentes}
\item
  \textbf{Reconstruye archivo} en destino
\end{enumerate}

\textbf{Resultado:} Transferencias ultra-rápidas, especialmente con
archivos grandes parcialmente modificados.

\subsection{Terminología}\label{terminologuxeda}

\textbf{Origen (Source):}

\begin{itemize}
\tightlist
\item
  El directorio/archivo a copiar
\item
  Puede ser local o remoto
\end{itemize}

\textbf{Destino (Destination):}

\begin{itemize}
\tightlist
\item
  Donde se copiarán los archivos
\item
  Puede ser local o remoto
\end{itemize}

\textbf{Trailing Slash (\texttt{/}):}

\begin{itemize}
\tightlist
\item
  \textbf{CON slash final:} \texttt{origen/} → copia CONTENIDO del
  directorio
\item
  \textbf{SIN slash final:} \texttt{origen} → copia el DIRECTORIO mismo
\end{itemize}

\begin{Highlighting}[]

\CommentTok{\# Copia: origen/archivo1 → destino/origen/archivo1}

\section{Backup Incremental}\label{backup-incremental}

\subsection{Concepto de Backup
Incremental}\label{concepto-de-backup-incremental}

\textbf{Backup completo:} Primera copia completa\\
\textbf{Backup incremental:} Solo cambios desde el último backup\\
\textbf{Ventaja:} Ahorra espacio y tiempo

\subsection{Método 1: Hard Links
(Recomendado)}\label{muxe9todo-1-hard-links-recomendado}

\begin{Shaded}
\begin{Highlighting}[]
\CommentTok{\# Primer backup (completo)}
\FunctionTok{rsync} \AttributeTok{{-}av} \AttributeTok{{-}{-}delete} \DataTypeTok{\textbackslash{}}
\NormalTok{  /origen/ }\DataTypeTok{\textbackslash{}}
\NormalTok{  /backup/2026{-}01{-}25/}

\CommentTok{\# Segundo backup (incremental con hard links)}
\FunctionTok{rsync} \AttributeTok{{-}av} \AttributeTok{{-}{-}delete} \DataTypeTok{\textbackslash{}}
  \AttributeTok{{-}{-}link{-}dest}\OperatorTok{=}\NormalTok{/backup/2026{-}01{-}25/ }\DataTypeTok{\textbackslash{}}
\NormalTok{  /origen/ }\DataTypeTok{\textbackslash{}}
\NormalTok{  /backup/2026{-}01{-}26/}

\CommentTok{\# Tercer backup}
\FunctionTok{rsync} \AttributeTok{{-}av} \AttributeTok{{-}{-}delete} \DataTypeTok{\textbackslash{}}
  \AttributeTok{{-}{-}link{-}dest}\OperatorTok{=}\NormalTok{/backup/2026{-}01{-}26/ }\DataTypeTok{\textbackslash{}}
\NormalTok{  /origen/ }\DataTypeTok{\textbackslash{}}
\NormalTok{  /backup/2026{-}01{-}27/}
\end{Highlighting}
\end{Shaded}

\textbf{Ventajas:}

\begin{itemize}
\tightlist
\item
  Cada backup parece completo
\item
  Archivos sin cambios son hard links (no duplican espacio)
\item
  Archivos modificados se copian completamente
\item
  Fácil restauración: elegir fecha y copiar
\end{itemize}

\textbf{Estructura resultante:}

\begin{verbatim}
/backup/
├── 2026-01-25/  (backup completo)
├── 2026-01-26/  (solo cambios, resto hard links)
└── 2026-01-27/  (solo cambios, resto hard links)
\end{verbatim}
